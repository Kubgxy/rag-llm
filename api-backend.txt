# --- ตัวแปรเก็บสถานะการประมวลผลไฟล์ (In-memory DB จำลอง) ---
# รูปแบบ: {"filename.pdf": {"status": "processing" | "completed" | "error", "summary": "", "mindmap": {}}}
doc_status = {}

# --- 1. ฟังก์ชัน OCR (เหมือนเดิม) ---
def extract_text_smart(file_path):
    text = ""
    try:
        reader = pypdf.PdfReader(file_path)
        for page in reader.pages:
            extracted = page.extract_text()
            if extracted: text += extracted + "\n"
    except: pass

    if len(text.strip()) < 100:
        try:
            images = convert_from_path(file_path, dpi=300)
            for img in images:
                text += pytesseract.image_to_string(img, lang='tha+eng') + "\n"
        except: pass
    return text

# --- 2. ฟังก์ชันทำงานเบื้องหลัง (Background Task) ---
def process_document_background(file_path: str, filename: str):
    try:
        # 1. สกัดข้อความ (ใช้เวลาตรงนี้นานสุด)
        extracted_text = extract_text_smart(file_path)
        
        # 2. บันทึกลง Database
        doc = Document(text=extracted_text, metadata={"file_name": filename})
        index = VectorStoreIndex.from_documents([doc], storage_context=storage_context)
        
        # 3. โหลด Chinda ชั่วคราวเพื่อทำ Summary & Mindmap
        llm_summarizer = Ollama(model="iapp/chinda-qwen3-4b", request_timeout=600.0, additional_kwargs={"num_ctx": 4096})
        engine = index.as_query_engine(llm=llm_summarizer, similarity_top_k=2)
        
        summary_res = engine.query("สรุปใจความสำคัญของเอกสารนี้เป็นภาษาไทย ความยาวไม่เกิน 5 bullet points")
        mindmap_res = engine.query("สกัดข้อมูลเพื่อทำ Mindmap ตอบกลับมาเป็น JSON format ที่มี keys คือ 'nodes' และ 'edges' ห้ามอธิบายเพิ่ม")
        
        # พยายามแกะ JSON ออกมาจากคำตอบของ AI (กันเหนียวกรณี AI ตอบ markdown ครอบมา)
        mindmap_json = {"nodes": [{"id": "1", "data": {"label": "แกะ Mindmap ไม่สำเร็จ"}}], "edges": []}
        json_match = re.search(r'\{.*\}', str(mindmap_res), re.DOTALL)
        if json_match:
            try: mindmap_json = json.loads(json_match.group(0))
            except: pass

        # 4. อัปเดตสถานะว่า "เสร็จแล้ว"
        doc_status[filename] = {
            "status": "completed",
            "summary": str(summary_res),
            "mindmap": mindmap_json
        }
        print(f"✅ ไฟล์ {filename} ประมวลผลเสร็จสมบูรณ์!")
    except Exception as e:
        doc_status[filename] = {"status": "error", "message": str(e)}
        print(f"❌ ประมวลผลไฟล์ {filename} พัง: {e}")

@app.post("/upload")
async def upload_document(background_tasks: BackgroundTasks, file: UploadFile = File(...)):
    """รับไฟล์ปุ๊บ คืนค่าปั๊บ แล้วเอาไปรันต่อเบื้องหลัง"""
    os.makedirs("uploaded_docs", exist_ok=True)
    file_path = f"uploaded_docs/{file.filename}"
    with open(file_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)
        
    # ตั้งสถานะเป็นกำลังทำงาน และสั่งรัน Background
    doc_status[file.filename] = {"status": "processing", "summary": "", "mindmap": {}}
    background_tasks.add_task(process_document_background, file_path, file.filename)
    
    return {"status": "success", "message": "อัปโหลดสำเร็จ กำลังเรียนรู้และทำสรุปเบื้องหลัง...", "filename": file.filename}

class SingleChatReq(BaseModel):

@app.post("/chat/single")

    try:
        index = VectorStoreIndex.from_vector_store(vector_store=vector_store)
        # โหลดโมเดลแบบ Dynamic (กิน RAM แค่ตอนเรียก)
        dynamic_llm = Ollama(model=req.model_name, request_timeout=600.0, additional_kwargs={"num_ctx": 4096})
        engine = index.as_query_engine(llm=dynamic_llm, similarity_top_k=2)

class CompareChatReq(BaseModel):
    query: str
    models: List[str]

@app.post("/chat/compare")
async def chat_compare(req: CompareChatReq):
    """โหลดโมเดลทีละตัวมาแข่งกันตอบ (รันทีละคิวเพื่อกัน RAM Colab ระเบิด)"""
    try:
        index = VectorStoreIndex.from_vector_store(vector_store=vector_store)
        results = {}
        # วนลูปให้คิดทีละตัว จะได้ไม่ดึง VRAM การ์ดจอพร้อมกันจนพัง
        for m in req.models:
            dynamic_llm = Ollama(model=m, request_timeout=600.0, additional_kwargs={"num_ctx": 4096})
            engine = index.as_query_engine(llm=dynamic_llm, similarity_top_k=2)
            ans = await asyncio.to_thread(engine.query, req.query)
            results[m] = str(ans)
            
        return {"query": req.query, "results": results}
    except Exception as e: return {"status": "error", "message": str(e)}
